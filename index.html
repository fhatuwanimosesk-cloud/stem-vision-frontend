# Drop-in replacement for the /process-image endpoint, using Claude instead of Gemini. # Assumes a Flask backend (the most common setup for a small Render-hosted app like # this one) — if your backend is actually Node/Express or something else, say so and # I'll rewrite this to match. # # Setup: # pip install anthropic # Set an ANTHROPIC_API_KEY environment variable on Render (Dashboard -> your # service -> Environment), the same way you set your Gemini key. # # The response shape below — {"solution_steps": [...]} / {"error": "..."} — matches # what your existing frontend already expects, so nothing in the HTML/JS needs to # change at all. import base64 from flask import Flask, request, jsonify from anthropic import Anthropic app = Flask(__name__) client = Anthropic() # reads ANTHROPIC_API_KEY from the environment automatically SYSTEM_PROMPT = """You are a patient, precise math and physics tutor. You will be shown \ a photo of a problem, along with a subject and academic level. Read the problem from \ the image and solve it step by step, at a level appropriate for the stated academic level. Respond with ONLY the inner HTML for the solution — no markdown code fences, no text \ outside the HTML. Use exactly these classes, which are already styled on the page: - One
restating the problem in your own words. - One
for each step of the working. - One
with the final answer, clearly stated. Write all mathematics using MathJax delimiters: \\( ... \\) for inline math, \\[ ... \\] \ for a displayed equation. Never use a bare $ for math.""" @app.route("/process-image", methods=["POST"]) def process_image(): file = request.files.get("file") subject = request.form.get("subject", "Advanced Mathematics") level = request.form.get("level", "Undergraduate") corrected_text = request.form.get("corrected_text", "") if not file: return jsonify({"error": "No image uploaded."}), 400 image_b64 = base64.standard_b64encode(file.read()).decode("utf-8") user_text = f"Subject: {subject}\nLevel: {level}" if corrected_text.strip(): user_text += f"\nStudent's note/correction: {corrected_text.strip()}" try: message = client.messages.create( model="claude-sonnet-5", max_tokens=2000, system=SYSTEM_PROMPT, messages=[{ "role": "user", "content": [ { "type": "image", "source": { "type": "base64", "media_type": file.mimetype or "image/jpeg", "data": image_b64, }, }, {"type": "text", "text": user_text}, ], }], ) html = "".join(block.text for block in message.content if block.type == "text") return jsonify({"solution_steps": [html]}) except Exception as e: # Surfaces as err.message in the frontend's NOTICE box. return jsonify({"error": str(e)}), 500
